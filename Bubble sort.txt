#The bubble sort
n = input("Enter the number of values: ")
n = int(n)
values = []

for i in range(1,n+1):
    value = input(" Enter the value number "+str(i)+":  ")
    value = int(value)
    values.append(value)

print(" Your sequence befor sorting is: "+str(values)+".")

for i in range(0,n-1):
	for j in range(0,n-1-i):
		if values[j] > values[j+1]:
			m = values[j]
			values[j] = values[j+1]
			values[j+1] = m

print(" Your sequence aftr sorting is: "+str(values)+".")
